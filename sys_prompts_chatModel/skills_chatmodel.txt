* orders: order_id (PK), customer_id (FK -> customers), order_date, order_status, shipping_city
* order_items: order_item_id (PK), order_id (FK -> orders), product_id (FK -> products), quantity, unit_price
* payments: payment_id (PK), order_id (FK -> orders), payment_date, payment_method, amount, payment_status
* deliveries: delivery_id (PK), order_id (FK -> orders), shipped_date, expected_date, delivered_date, delivery_status
