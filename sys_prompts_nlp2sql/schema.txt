# Database Schema Summary
Table customers {
  customer_id uuid [PK]
  full_name text
  email text
  city text
  customer_segment text
  signup_date date
}

Table suppliers {
  supplier_id uuid [PK]
  supplier_name text
  city text
  rating numeric
}

Table products {
  product_id uuid [PK]
  product_name text
  category text
  brand text
  supplier_id uuid [FK -> suppliers.supplier_id]
  cost_price numeric
  selling_price numeric
}

Table orders {
  order_id uuid [PK]
  customer_id uuid [FK -> customers.customer_id]
  order_date date
  order_status text [Enum: 'Pending', 'Processing', 'Shipped', 'Delivered', 'Cancelled']
  shipping_city text
}

Table order_items {
  order_item_id uuid [PK]
  order_id uuid [FK -> orders.order_id]
  product_id uuid [FK -> products.product_id]
  quantity integer
  unit_price numeric
}

Table payments {
  payment_id uuid [PK]
  order_id uuid [FK -> orders.order_id]
  payment_date date
  payment_method text [Enum: 'Credit Card', 'PayPal', 'Bank Transfer']
  amount numeric
  payment_status text [Enum: 'Success', 'Failed', 'Refunded']
}

Table deliveries {
  delivery_id uuid [PK]
  order_id uuid [FK -> orders.order_id]
  shipped_date date
  expected_date date
  delivered_date date
  delivery_status text [Enum: 'In Transit', 'Delivered', 'Returned', 'Failed']
}
